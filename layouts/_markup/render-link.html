{{- /* The rules and legal pages are written to read on GitHub too, so they link with relative .md paths. Resolve those to site pages; a relative link that doesn't resolve fails the build. */ -}} {{- $u := urls.Parse .Destination -}} {{- $href := .Destination -}} {{- $external := false -}} {{- if $u.IsAbs -}} {{- $external = true -}} {{- else if and $u.Path (not (strings.HasPrefix $u.Path "/")) -}} {{- $page := .PageInner -}} {{- $rel := strings.TrimPrefix "/" (path.Clean (path.Join $page.File.Dir $u.Path)) -}} {{- $frag := cond (ne $u.Fragment "") (printf "#%s" $u.Fragment) "" -}} {{- $key := strings.TrimSuffix "/_index" (strings.TrimSuffix ".md" $rel) -}} {{- with site.GetPage (printf "/%s" $key) -}} {{- $href = printf "%s%s" .RelPermalink $frag -}} {{- else -}} {{- errorf "%s: link %q does not resolve to a page" $page.File.Path .Destination -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }} {{- /* chomp */ -}}
