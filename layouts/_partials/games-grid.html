{{- /* The games grid, from data/games.yaml. Art is each game's Steam library hero, which has no logo. Newer games keep their art at hashed URLs, so the store API (IStoreBrowseService, no key needed) is asked where it is, once for all games. Everything is fetched at build time and served from this site, so the CSP stays img-src 'self'. A failed fetch fails the build rather than shipping a broken card. */ -}} {{- $games := hugo.Data.games -}} {{- $ids := slice -}} {{- range $games }}{{ with .steam_appid }}{{ $ids = $ids | append (dict "appid" .) }}{{ end }}{{ end -}} {{- $assets := dict -}} {{- if $ids -}} {{- $q := dict "ids" $ids "context" (dict "language" "english" "country_code" "US") "data_request" (dict "include_assets" true) | jsonify -}} {{- $api := printf "https://api.steampowered.com/IStoreBrowseService/GetItems/v1/?input_json=%s" ($q | urlquery) -}} {{- with try (resources.GetRemote $api) -}} {{- with .Err }}{{ errorf "games grid: Steam store API: %s" . }}{{ end -}} {{- with .Value -}} {{- range (. | transform.Unmarshal).response.store_items -}} {{- if and .assets .assets.library_hero -}} {{- $path := replace .assets.asset_url_format "${FILENAME}" .assets.library_hero -}} {{- $assets = merge $assets (dict (string .appid) (printf "https://shared.akamai.steamstatic.com/store_item_assets/%s" $path)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
